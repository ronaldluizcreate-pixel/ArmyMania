O brilho é surreal, super recomendo a ArmieMania." }, { nome: "Lucas M.", bias: "Suga", nota: 5, comentario: "Comprei a Box Surpresa e veio tudo muito bem embalado com cheirinho ótimo. Atendimento nota 10!" }, { nome: "Beatriz R.", bias: "Jimin", nota: 5, comentario: "Melhor loja de brindes BTS! Já fiz 3 pedidos e sempre chega antes do prazo." } ]; onSnapshot(collection(db, "depoimentos"), (snap) => { const reviews = snap.empty ? demoReviews : snap.docs.map(d => d.data()); const grid = document.getElementById('reviewsGrid'); grid.innerHTML = reviews.map(r => `

${r.nome}

Bias: ${r.bias || 'OT7'}
${'⭐'.repeat(r.nota || 5)}

"${r.comentario}"

`).join(''); }); window.openReviewModal = () => { if (!currentUser) { openAuthModal(); return alert("Entre na sua conta para deixar seu feedback!"); } const modal = document.getElementById('reviewModal'); modal.classList.remove('hidden'); modal.classList.add('flex'); }; window.closeReviewModal = () => { const modal = document.getElementById('reviewModal'); modal.classList.add('hidden'); modal.classList.remove('flex'); }; window.submitReview = async (e) => { e.preventDefault(); const nota = Number(document.getElementById('reviewRating').value); const comentario = document.getElementById('reviewText').value; try { await addDoc(collection(db, "depoimentos"), { clienteId: currentUser.uid, nome: currentUserData?.nome || currentUser.email, bias: currentUserData?.bias || "OT7", nota, comentario, criadoEm: serverTimestamp() }); document.getElementById('reviewText').value = ''; closeReviewModal(); alert("Obrigado pelo seu feedback!"); } catch (err) { alert("Erro ao enviar: " + err.message); } };
